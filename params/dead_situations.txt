# Lines:   sign / confirm / end  =  日本語 | English | expression        ('#' starts a comment)
# Columns: see possession/v2/situation_features.py (or the end of the report).
# Helpers: lasting(x, s)  recent(x, s)  ahead(x, s)  abs(x)   and / or / not
# Every list should have 5 signs; `need` of them (2) must be true at the same time.

[settings]
need = 2              # 5つのうちいくつ満たせばよいか / how many of the 5 signs must hold
start_hold_s = 1.0    # 開始サインが続く時間 / start signs must hold this long

[start | 開始のサイン]
# D の始まりを「はっきりプレー中だった最後の瞬間」の直後まで戻す（最大 start_back_max_s 秒）
back_to = 速い選手が3人以上いた最後の時点 | last moment 3+ players ran fast | n_fast >= 3
sign = 2秒間ほぼ全員が歩いている | nearly everybody has walked for 2 s | lasting(frac_walk >= 0.8, 2)
sign = 2秒間 2番目に速い選手も遅い | even the 2nd fastest has been slow for 2 s | lasting(sp2 < 2.5, 2)
sign = 直前まで速かったのに今は遅い | fast just before, slow now | recent(sp2 > 4.5, 4) and sp2 < 2.5
sign = 減速する選手が続く | players keep slowing down | lasting(n_decel >= 1, 1)
sign = セットプレー/ゴールのタグが近い | a set-piece or goal tag is near | tag_sp_to_start_s <= 12 or tag_sp_since_start_s <= 2 or tag_since_goal_s <= 5
